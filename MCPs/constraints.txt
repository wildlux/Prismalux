# MCPs/constraints.txt — versioni allineate per i pacchetti pip condivisi tra più MCP
#
# Non è un requirements.txt: non installa nulla da solo, vincola solo la
# versione di un pacchetto che un requirements.txt ha GIA' richiesto (pip
# risolve l'intersezione tra le due specifiche). Serve perché tutti i MCP
# condividono lo stesso venv (McpManagerPage::venvPython(), un solo
# ambiente per l'intero progetto) — due MCP con versioni diverse dello
# stesso pacchetto in requirements.txt separati derivavano silenziosamente
# verso qualunque versione installata per ultima, senza avviso.
#
# Uso in ogni requirements.txt (percorso relativo al file stesso):
#   -c ../constraints.txt
#
# Per aggiungere un nuovo pacchetto condiviso: pinnarlo qui UNA volta,
# lasciare il requirements.txt del singolo MCP senza numero di versione
# (es. "Pillow" invece di "Pillow==12.3.0").

# Pillow — richiesto da rdkit_mcp (immagini molecole) e stable_diffusion_local
# (pipeline diffusers/torch, versione testata insieme al resto dello stack
# 2026-05-15). Stessa versione per entrambi: evita che l'installazione
# dell'uno sovrascriva silenziosamente quella testata dall'altro nel venv
# condiviso.
Pillow==12.3.0

# requests — richiesto da gns3_mcp. Centralizzato qui così un futuro MCP
# che aggiunge "requests" al proprio requirements.txt eredita la stessa
# versione testata invece di sceglierne una propria.
requests==2.34.2
